Vite  (means quick in English)

  A build too that aims to provide a faster and leaner development experience for modern
  web projects.


  React installation using Vite

  using Vite
    $ npm create vite@latest

  Updateing Vite
    $ npm install vite@latest --save-dev  

  Running Vite server:

    $ npm run dev

  - Change Server port number (to become port number 3000)
  - Add mkcert plugin for https connection
  - Set a default path for your Vite project using 'path'
    
    Go to file cite.config.ts

    import { defineConfig } from "vite";
    import react from "@vitejs/plugin-react";
    import mkcert from "vite-plugin-mkcert";
    import tailwindcss from "@tailwindcss/vite";
    import path from "path";

    // https://vite.dev/config/
    export default defineConfig({
      server: {
        port: 3000,
      },
      plugins: [react(), mkcert(), tailwindcss()],
      resolve: {
        alias: {
          "@": path.resolve(__dirname, "./src"),
        },
      },
    });
